## Executive Summary

The KPU simulator has **functional C++ execution infrastructure** that computes correct DNN outputs. However, the **benchmarking and validation foundation** (v0.3.0) was never completed before advancing to v0.8.0. This gap means we cannot verify that our C++ simulation accurately models the target hardware's performance characteristics.

### Impact

| Risk | Severity | Description |
|------|----------|-------------|
| Silent performance bugs | HIGH | C++ may compute values but with incorrect timing |
| No regression detection | HIGH | Performance regressions go unnoticed |
| Unvalidated roofline | MEDIUM | Don't know if ops are memory/compute bound correctly |
| Missing CI guardrails | MEDIUM | No automated performance verification |

```cpp
// tests/benchmarks/matmul_sweep.cpp
#include <sw/kpu/models/behavioral/compute/compute_fabric.hpp>
#include <sw/xue/event_collector.hpp>
#include <chrono>
#include <vector>

struct BenchmarkResult {
    uint32_t M, N, K;
    uint64_t flops;
    double elapsed_ms;
    double gflops;
    double efficiency;  // vs theoretical peak
};

std::vector<BenchmarkResult> run_matmul_sweep() {
    std::vector<uint32_t> sizes = {64, 128, 256, 512, 1024, 2048, 4096, 8192};
    std::vector<BenchmarkResult> results;

    ComputeFabricConfig config;
    config.macs_per_cycle = 256;  // 16x16 PE array
    BehavioralComputeFabric fabric(config, 0);

    for (uint32_t size : sizes) {
        // Allocate matrices
        std::vector<float> A(size * size), B(size * size), C(size * size);

        // Reset XUE
        sw::xue::xue().reset();

        // Time execution
        auto start = std::chrono::high_resolution_clock::now();

        MatMulDescriptor desc{size, size, size, 4, false};
        fabric.submit_matmul(desc, A.data(), B.data(), C.data(), nullptr);

        auto end = std::chrono::high_resolution_clock::now();
        double elapsed_ms = std::chrono::duration<double, std::milli>(end - start).count();

        // Calculate metrics
        uint64_t flops = 2ULL * size * size * size;
        double gflops = flops / (elapsed_ms * 1e6);
        double peak_gflops = config.macs_per_cycle * 2 * 1000.0;  // At 1GHz
        double efficiency = gflops / peak_gflops;

        results.push_back({size, size, size, flops, elapsed_ms, gflops, efficiency});
    }

    return results;
}
```

**Success Criteria:**
- [ ] Benchmark runs sizes 64 to 8192
- [ ] Reports GFLOPS for each size
- [ ] Large sizes achieve >80% of peak
- [ ] Results exportable to CSV/JSON

```python
# tools/benchmark/roofline_generator.py
import matplotlib.pyplot as plt
import numpy as np
import kpu

def generate_roofline(results: list[dict], output_path: str):
    """Generate roofline plot from benchmark results."""

    # Hardware parameters (from KPU config)
    peak_gflops = 512  # Example: 256 MACs/cycle * 2 * 1GHz
    peak_bandwidth_gb = 100  # Example: LPDDR5

    # Ridge point
    ridge_point = peak_gflops / peak_bandwidth_gb

    # Plot roofline
    fig, ax = plt.subplots(figsize=(10, 6))

    # Roofline ceiling
    ai_range = np.logspace(-2, 3, 100)
    performance = np.minimum(peak_gflops, ai_range * peak_bandwidth_gb)
    ax.loglog(ai_range, performance, 'b-', linewidth=2, label='Roofline')

    # Plot operations
    for r in results:
        ai = r['arithmetic_intensity']
        gflops = r['achieved_gflops']
        ax.scatter(ai, gflops, s=100, label=r['name'])

    # Ridge point annotation
    ax.axvline(x=ridge_point, color='r', linestyle='--', alpha=0.5)
    ax.annotate(f'Ridge Point: {ridge_point:.1f}',
                xy=(ridge_point, peak_gflops/2), fontsize=10)

    ax.set_xlabel('Arithmetic Intensity (FLOPs/Byte)')
    ax.set_ylabel('Performance (GFLOPS)')
    ax.set_title('KPU Roofline Model')
    ax.legend()
    ax.grid(True, alpha=0.3)

    plt.savefig(output_path, dpi=150, bbox_inches='tight')
    print(f"Saved roofline plot to {output_path}")
```

```cpp
// tests/validation/analytical_matmul.cpp
#include <gtest/gtest.h>
#include <sw/kpu/models/behavioral/compute/compute_fabric.hpp>
#include <sw/xue/event_collector.hpp>

class AnalyticalMatmulTest : public ::testing::Test {
protected:
    void SetUp() override {
        // Known configuration: 16x16 PE array, 256 MACs/cycle
        config_.macs_per_cycle = 256;
        fabric_ = std::make_unique<BehavioralComputeFabric>(config_, 0);
        sw::xue::xue().reset();
    }

    ComputeFabricConfig config_;
    std::unique_ptr<BehavioralComputeFabric> fabric_;
};

TEST_F(AnalyticalMatmulTest, SingleTile16x16) {
    // ANALYTICAL SOLUTION:
    // 16x16 matmul = 16*16*16 = 4096 MACs = 8192 FLOPs
    // With 256 MACs/cycle: latency = 4096/256 = 16 cycles

    constexpr uint32_t M = 16, N = 16, K = 16;
    constexpr uint64_t EXPECTED_MACS = M * N * K;  // 4096
    constexpr uint64_t EXPECTED_FLOPS = EXPECTED_MACS * 2;  // 8192
    constexpr uint32_t EXPECTED_CYCLES = EXPECTED_MACS / 256;  // 16

    std::vector<float> A(M * K, 1.0f), B(K * N, 1.0f), C(M * N, 0.0f);

    MatMulDescriptor desc{M, N, K, 4, false};
    fabric_->submit_matmul(desc, A.data(), B.data(), C.data(), nullptr);

    // Verify XUE events
    auto summary = sw::xue::xue().get_summary();

    EXPECT_EQ(summary.total_flops, EXPECTED_FLOPS)
        << "FLOPs mismatch: expected " << EXPECTED_FLOPS;

    EXPECT_EQ(summary.matmul_count, 1)
        << "Should record exactly 1 matmul event";

    // Verify computed result (each element = K = 16)
    for (uint32_t i = 0; i < M * N; ++i) {
        EXPECT_FLOAT_EQ(C[i], static_cast<float>(K));
    }
}

TEST_F(AnalyticalMatmulTest, FourTiles32x32) {
    // ANALYTICAL SOLUTION:
    // 32x32 matmul tiled as 4 x (16x16) = 4 tiles
    // Each tile: 4096 MACs, Total: 16384 MACs = 32768 FLOPs
    // Ideal latency: 16384/256 = 64 cycles
    // With perfect pipelining: ~64 cycles
    // Without pipelining: 4 * 16 = 64 cycles

    constexpr uint32_t M = 32, N = 32, K = 32;
    constexpr uint64_t EXPECTED_MACS = M * N * K;  // 32768
    constexpr uint64_t EXPECTED_FLOPS = EXPECTED_MACS * 2;  // 65536

    std::vector<float> A(M * K, 1.0f), B(K * N, 1.0f), C(M * N, 0.0f);

    MatMulDescriptor desc{M, N, K, 4, false};
    fabric_->submit_matmul(desc, A.data(), B.data(), C.data(), nullptr);

    auto summary = sw::xue::xue().get_summary();

    EXPECT_EQ(summary.total_flops, EXPECTED_FLOPS);

    // Compute efficiency: actual cycles vs ideal
    double efficiency = static_cast<double>(EXPECTED_MACS) /
                       (summary.total_cycles * config_.macs_per_cycle);

    EXPECT_GE(efficiency, 0.8) << "Efficiency should be >80%";
}

TEST_F(AnalyticalMatmulTest, SixteenTiles64x64) {
    constexpr uint32_t M = 64, N = 64, K = 64;
    constexpr uint64_t EXPECTED_MACS = M * N * K;  // 262144
    constexpr uint64_t EXPECTED_FLOPS = EXPECTED_MACS * 2;  // 524288

    std::vector<float> A(M * K, 1.0f), B(K * N, 1.0f), C(M * N, 0.0f);

    MatMulDescriptor desc{M, N, K, 4, false};
    fabric_->submit_matmul(desc, A.data(), B.data(), C.data(), nullptr);

    auto summary = sw::xue::xue().get_summary();

}

TEST_F(AnalyticalMatmulTest, SixtyFourTiles128x128) {
    constexpr uint32_t M = 128, N = 128, K = 128;
    constexpr uint64_t EXPECTED_MACS = M * N * K;  // 2097152
    constexpr uint64_t EXPECTED_FLOPS = EXPECTED_MACS * 2;  // 4194304

    std::vector<float> A(M * K, 1.0f), B(K * N, 1.0f), C(M * N, 0.0f);

    MatMulDescriptor desc{M, N, K, 4, false};
    fabric_->submit_matmul(desc, A.data(), B.data(), C.data(), nullptr);

    auto summary = sw::xue::xue().get_summary();

    EXPECT_EQ(summary.total_flops, EXPECTED_FLOPS);

    // For large matmuls, efficiency should approach 100%
```

```python
# python/tests/test_xue_accuracy.py
import pytest
import numpy as np
import kpu

class TestXUEPredictionAccuracy:
    """Validate XUE predictions are within 10% of actual simulation."""

    def setup_method(self):
        kpu.reset_xue_counters()

    def test_matmul_prediction_accuracy(self):
        """XUE roofline prediction should be within 10% for matmul."""
        # Run a compute-bound matmul
        M, N, K = 1024, 1024, 1024
        A = np.random.randn(M, K).astype(np.float32)
        B = np.random.randn(K, N).astype(np.float32)

        result = kpu.matmul(A, B)

        analysis = kpu.get_operational_analysis()

        # Predicted vs actual
        predicted_gflops = analysis['predicted_gflops']
        actual_gflops = analysis['actual_gflops']

        error_percent = abs(predicted_gflops - actual_gflops) / actual_gflops * 100

        assert error_percent < 10, \
            f"Prediction error {error_percent:.1f}% exceeds 10% threshold"

    def test_conv2d_prediction_accuracy(self):
        """XUE roofline prediction should be within 10% for conv2d."""
        # Run a compute-bound conv2d
        x = np.random.randn(1, 64, 56, 56).astype(np.float32)
        w = np.random.randn(128, 64, 3, 3).astype(np.float32)

        result = kpu.conv2d(x, w, padding=1)

        analysis = kpu.get_operational_analysis()

        error_percent = analysis.get('gflops_error_percent', 0)

        assert error_percent < 10, \
            f"Conv2d prediction error {error_percent:.1f}% exceeds 10% threshold"

    def test_memory_bound_prediction(self):
        """XUE should correctly identify memory-bound operations."""
        # Run a memory-bound operation (small matmul, low arithmetic intensity)
        M, N, K = 32, 32, 32
        A = np.random.randn(M, K).astype(np.float32)
        B = np.random.randn(K, N).astype(np.float32)

        result = kpu.matmul(A, B)

        analysis = kpu.get_operational_analysis()

        # Small matmul should be memory-bound
        ai = analysis['arithmetic_intensity']
        ridge_point = analysis['hardware']['ridge_point_dram']

        # If AI < ridge point, should be memory-bound
        if ai < ridge_point:
            assert analysis.get('memory_bound', False), \
                "Small matmul should be identified as memory-bound"

    def test_compute_bound_prediction(self):
        """XUE should correctly identify compute-bound operations."""
        # Run a compute-bound operation (large matmul, high arithmetic intensity)
        M, N, K = 4096, 4096, 4096
        A = np.random.randn(M, K).astype(np.float32)
        B = np.random.randn(K, N).astype(np.float32)

        result = kpu.matmul(A, B)

        analysis = kpu.get_operational_analysis()

        ai = analysis['arithmetic_intensity']
        ridge_point = analysis['hardware']['ridge_point_dram']

        # Large matmul should be compute-bound
        if ai > ridge_point:
            assert analysis.get('compute_bound', True), \
                "Large matmul should be identified as compute-bound"
```

## Appendix: XUE Event Summary Structure

For reference, the XUE summary returned by `sw::xue::xue().get_summary()` contains:

```cpp
struct XUESummary {
    // Event counts
    uint64_t matmul_count;
    uint64_t conv_count;
    uint64_t elementwise_count;
    uint64_t pool_count;
    uint64_t softmax_count;
    uint64_t layernorm_count;

    // Aggregate metrics
    uint64_t total_flops;
    uint64_t total_bytes_moved;
    uint64_t total_cycles;

    // Derived metrics
    double arithmetic_intensity;  // FLOPs / Bytes
    double achieved_gflops;       // FLOPs / (cycles * cycle_time)
};
```

This is the data we need to validate against analytical solutions.
